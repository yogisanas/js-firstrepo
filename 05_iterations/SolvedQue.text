#1. Use filter() and map() to get the names of students who scored more than 50 marks.

const students = [
    { name: "Amit", marks: 35 },
    { name: "Priya", marks: 80 },
    { name: "Rahul", marks: 25 },
    { name: "Neha", marks: 90 },
    { name: "Sneha", marks: 60 }
];

const val = students
            .filter((item) => item.marks > 50)
            .map((item) => item.name)

console.log(val);

--[ 'Priya', 'Neha', 'Sneha' ]


#2. Find the total marks of students who scored more than 50, using filter() and reduce().

const students = [
    { name: "Amit", marks: 35 },
    { name: "Priya", marks: 80 },
    { name: "Rahul", marks: 25 },
    { name: "Neha", marks: 90 },
    { name: "Sneha", marks: 60 }
];

const val = students
            .filter((item) => item.marks > 50)
            .reduce((acc, item) => acc + item.marks, 0)

console.log(val);

--230


#3. calculate the total price of all Electronics products costing more than ₹5,000.

//#. Find the maximum number.

    if(max > item){
        return max
    } else{
        return item
    }
}, nums[0])  --can take 0 also

console.log(val);

//#. Find the longest string in an array using.

const names = ["Amit", "Priya", "Rahul", "Alexander"];

const result = names.reduce((longest, name) => {
    return name.length > longest.length ? name : longest;
});

//#. Count how many students passed, given that passing marks are 40.

const students = [
    { name: "Amit", marks: 35 },
    { name: "Priya", marks: 80 },
    { name: "Rahul", marks: 25 },
    { name: "Neha", marks: 90 },
    { name: "Sneha", marks: 40 }
];

// const result = students.filter(item => item.marks >= 40).length;

const result = students.reduce(
    (count, item) => item.marks >= 40 ? count + 1 : count,
    0
);

console.log(result);  

--3
